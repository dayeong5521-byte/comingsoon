async function startHunt() { var kw = document.getElementById('mainSearch').value.trim(); if (!kw) { document.getElementById('mainSearch').focus(); return; } var btn = document.getElementById('huntBtn'); btn.disabled = true; btn.textContent = '탐색 중...'; allItems = []; rebuildNav(); showSkeleton(); setStatus("'" + kw + "' 수색 중...", true); try { var res = await fetch('/api/hunt', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ keyword: kw }) }); if (!res.ok) { var err = await res.json(); throw new Error(err.error || '서버 오류 ' + res.status); } // SSE 스트림 읽기 var reader = res.body.getReader(); var decoder = new TextDecoder(); var buf = ''; while (true) { var chunk = await reader.read(); if (chunk.done) break; buf += decoder.decode(chunk.value, { stream: true }); var parts = buf.split('\n\n'); buf = parts.pop(); for (var i = 0; i < parts.length; i++) { var line = parts[i]; if (!line.startsWith('data: ')) continue; try { var ev = JSON.parse(line.slice(6)); if (ev.type === 'status') setStatus(ev.message, true); else if (ev.type === 'item') { allItems.push(ev.data); rebuildNav(); renderItems(); } else if (ev.type === 'done') setStatus('탐색 완료! ' + ev.total + '개 시그널 확보.', false); else if (ev.type === 'error') throw new Error(ev.message); } catch(e) { if (e.message && !e.message.includes('JSON')) throw e; } } } if (!allItems.length) document.getElementById('content').innerHTML = '
결과를 찾지 못했습니다
다른 키워드로 시도해보세요.
'; } catch(err) { setStatus('', false); document.getElementById('content').innerHTML = '
오류 발생
' + err.message + '
'; } finally { btn.disabled = false; btn.innerHTML = ' 발견하기'; } }
